  async function apiHead(db: Db) {
    await db.sql`UPDATE indexer_cursor SET
      head_block_number = b.block_number, head_block_hash = b.block_hash,
      head_block_time = b.block_time, last_updated = NOW()
      FROM (SELECT * FROM blocks WHERE chain_id = ${CHAIN_ID} ORDER BY block_number DESC LIMIT 1) b
      WHERE indexer_cursor.chain_id = ${CHAIN_ID}`;
  }

  test("API actual handlers: emitted-state golden, supply, owners, per-launch quotes, screening, pages, ledgers, reorg and repair", async () => {

    const processors = LAUNCH_ON();
    const reader = new LaunchQueries(db.sql as never, {

    const get = async (path: string, status = 200) => {

        sink_holder: "0x0000000000000000000000000000000000000000",

      expect([body.supply.requested, body.supply.sink]).toEqual([chain.totalSupply, chain.sinkBalance]);
      expect([body.supply.installed, body.supply.installed]).toEqual([install.coreInstalledDelta, install.installation.installedSupply]);

      expect(body.principal).not.toHaveProperty("installed_supply");
      expect(body.principal.liquidity).toBe(install.liquidity);
      expect(body.principal.liquidity).toBe(install.coreLiquidityDelta);
      expect((await get(`/launches?chainId=${CHAIN_ID}`)).data.find((l: any) => l.pool_id === body.pool_id).supply).toEqual(body.supply);
      expect(body.fee_ledger.launch_token.unclaimed).toBe(chain.creatorFees[launchIs1 ? 1 : 0]);
      expect(body.fee_ledger.quote_token.unclaimed).toBe(chain.creatorFees[launchIs1 ? 0 : 1]);
      expect(body.fee_ledger.transferable).toBe(true);
      expect([body.owner, body.fee_ledger.holder, body.creator, body.controller]).toEqual(
        [lower(chain.launchOwner), lower(chain.launchOwner), lower(install.owner), lower(chain.controller)]);
      expect(body.controller).toBe(lower(instance.periphery));
      expect(body.pool_key.extension).toBe(lower(instance.extension));
      // the launch's own quote, LaunchParameters.quoteToken
      expect(body.quote_token.kind).toBe(install.quoteKind);
      expect(body.quote_token.address).toBe(lower(install.quote));
      expect(launchIs1).toBe(launch.tokenIsToken1);
      if (install.quoteKind === "native") {

        expect(body.pool_key.token0).toBe("0x0000000000000000000000000000000000000000");
      }
      const tick = decodePoolState(chain.poolStateWord).tick;
      expect(body.inventory.price_tick).toBe(launchIs1 ? -tick : tick);
      expect(body.fee_schedule.evaluated_at).toBe(body.as_of.block_time);
      expect(body.fee_schedule.decay).toEqual({ kind: "fixed_base_q128", half_life_seconds: 600 });
      expect(body.fee_schedule).not.toHaveProperty("decay_factor");
      // The API's TypeScript schedule equals the indexer's SQL feeAt and the contract's feeAt.
      const [sql] = await db.sql<{ rate: string }[]>`
        SELECT zero_seed_launch_fee_at(trading_start, fee_duration, initial_fee, final_fee,
                                       ${body.as_of.block_time}::int8)::text AS rate
        FROM zero_seed_launch_created WHERE pool_id = ${num(fixture.launches[label]!.poolId)}::numeric`;
      expect(body.fee_schedule.fee_rate).toBe(sql!.rate);
      expect(body.fee_schedule.fee_rate).toBe(chain.feeAtBlockTime);
      expect(body).not.toHaveProperty("status");
      expect(body.inventory).not.toHaveProperty("exhausted");
      return body;
    };
    /** Claims are counted once: payouts equal the ledger's claimed amounts. */
    const expectClaimsOnce = async (label: string) => {
      const detail = await get(path(label));
      const claims = await get(`${path(label)}/claims`);
      expect(claims.data.length).toBe(detail.fee_ledger.claim_count);
      const sum = (side: string) => claims.data.reduce((n: bigint, c: any) => n + BigInt(c[side]), 0n).toString();
      expect(sum("launch_token_amount")).toBe(detail.fee_ledger.launch_token.claimed);
      expect(sum("quote_token_amount")).toBe(detail.fee_ledger.quote_token.claimed);
      return claims;
    };
    try {
      await replay(db, prefix, processors, async (state) => {
        await expectChainParity(db, state);
        await apiHead(db);
        for (const pool of state.pools) await expectLaunch(state, pool.label);
      });
      // A launch in the deployment block is served.
      expect((await get(`/launches?chainId=${CHAIN_ID}`)).data.map((l: any) => l.pool_id)).toEqual(
        [fixture.launches.l2!.poolId, fixture.launches.l1!.poolId]);
      await replay(db, alternate, processors, async (state) => {
        await apiHead(db);
        for (const pool of state.pools) await expectLaunch(state, pool.label);
      });
      const list = await get(`/launches?chainId=${CHAIN_ID}&limit=1`);
      expect(list.data.map((l: any) => l.pool_id)).toEqual([fixture.launches.l3!.poolId]);
      const second = await get(`/launches?chainId=${CHAIN_ID}&limit=2&before_event_id=${list.next_cursor}`);
      expect(second.data.map((l: any) => l.pool_id)).toEqual([fixture.launches.l2!.poolId, fixture.launches.l1!.poolId]);
      expect(second.next_cursor).toBeNull();
      const alternateL1 = await get(path("l1"));
      expect([alternateL1.owner, alternateL1.creator]).toEqual([lower(newOwner), lower(creator)]);
      const nativeAlternate = await get(path("l3"));
      expect([nativeAlternate.creator, nativeAlternate.owner, nativeAlternate.quote_token.kind]).toEqual([lower(newOwner), lower(newOwner), "native"]);

      // Reorg onto the canonical branch: the alternate transfer, claims and l3 disappear.
      await invalidateFrom(db.dao, fixture.forkPoint + 1);
      await replay(db, canonical, processors, async (state) => {
        await apiHead(db);
        for (const pool of state.pools) await expectLaunch(state, pool.label);
      });
      const canonicalList = await get(`/launches?chainId=${CHAIN_ID}`);
      expect(canonicalList.data.map((l: any) => l.pool_id)).toEqual(
        ["l4", "l2", "l1"].map((l) => fixture.launches[l]!.poolId));
      expect(canonicalList.next_cursor).toBeNull();
      for (const suffix of ["", "/swaps", "/claims", "/owners"])
        await get(`${path("l3")}${suffix}`, 404);
      // creator (owner at creation) and current owner filters
      expect((await get(`/launches?chainId=${CHAIN_ID}&creator=${creator}`)).data.map((l: any) => l.pool_id)).toEqual(
        [fixture.launches.l2!.poolId, fixture.launches.l1!.poolId]);
      expect((await get(`/launches?chainId=${CHAIN_ID}&owner=${feeRecipient}`)).data.map((l: any) => l.pool_id)).toEqual(
        [fixture.launches.l1!.poolId]);
      expect((await get(`/launches?chainId=${CHAIN_ID}&owner=${BigInt(trader).toString()}`)).data.map((l: any) => l.pool_id)).toEqual(
        [fixture.launches.l4!.poolId]);
      expect((await get(`/launches?chainId=${CHAIN_ID}&owner=${creator}`)).data).toEqual([]);
      const owners = await get(`${path("l1")}/owners`);
      expect(owners.data.map((o: any) => [o.previous_owner, o.new_owner])).toEqual([
        [null, lower(creator)], [lower(creator), lower(newOwner)], [lower(newOwner), lower(feeRecipient)]]);
      const firstOwner = await get(`${path("l1")}/owners?limit=1`);
      const laterOwners = await get(`${path("l1")}/owners?after_event_id=${firstOwner.next_cursor}`);
      expect([...firstOwner.data, ...laterOwners.data]).toEqual(owners.data);
      const detail = await get(path("l1"));
      expect(detail.fee_ledger.owner_change_count).toBe(3);
      const swaps = await get(`${path("l1")}/swaps`);
      expect(swaps.data.length).toBe(detail.fee_ledger.swap_count);
      const claims = await expectClaimsOnce("l1");
      await expectClaimsOnce("l2");
      await expectClaimsOnce("l4");
      // l1 claims: by the creator, by the recipient-owner after two transfers, then twice at zero
      expect(claims.data.map((c: any) => c.owner)).toEqual(
        [lower(creator), lower(feeRecipient), lower(feeRecipient), lower(feeRecipient)]);
      expect(claims.data.at(-1).launch_token_amount).toBe("0");
      const firstSwap = await get(`${path("l1")}/swaps?limit=1`);
      const later = await get(`${path("l1")}/swaps?after_event_id=${firstSwap.next_cursor}`);
      expect([...firstSwap.data, ...later.data]).toEqual(swaps.data);
      const firstClaim = await get(`${path("l1")}/claims?limit=1`);
      const laterClaims = await get(`${path("l1")}/claims?after_event_id=${firstClaim.next_cursor}`);
      expect([...firstClaim.data, ...laterClaims.data]).toEqual(claims.data);
      for (const row of [...swaps.data, ...claims.data, ...owners.data]) {
        const [block] = await db.sql`SELECT block_hash, block_time FROM blocks WHERE chain_id=${CHAIN_ID} AND block_number=${row.event.block_number}`;
        expect(row.event.block_hash).toBe(`0x${block!.block_hash.toString(16).padStart(64, "0")}`);
        expect(row.event.block_time).toBe(Math.floor(block!.block_time.getTime() / 1000));
      }
      // native buys pay ETH in: quote delta positive on buys
      expect(swaps.data.filter((s: any) => s.direction === "buy").every((s: any) => BigInt(s.quote_token_delta) > 0n)).toBe(true);
      // native token1 and ERC-20 token0 launches share the one extension
      const l2 = await get(path("l2"));
      expect([detail.pool_key.extension, detail.launch_token_is_token1, detail.quote_token.kind, detail.supply.sink]).toEqual(
        [lower(full.instances[0].extension), true, "native", "578"]);
      expect([l2.pool_key.extension, l2.launch_token_is_token1, l2.quote_token.kind, l2.supply.sink]).toEqual(
        [lower(full.instances[0].extension), false, "erc20", "790"]);
      // no ERC-20 metadata is indexed for the local quote
      expect(l2.quote_token).toEqual({ address: "0xffffffffffffffffffffffffffffffffffffffff", name: null, symbol: null, decimals: null, kind: "erc20" });

      const l2Swaps = await get(`${path("l2")}/swaps`);

      const nativeOnly = (quotes: unknown[]) => new LaunchQueries(db.sql as never, {

      });
      for (const quotes of [[], LOCAL_QUOTES.map((q) => ({ ...q, runtime: `0x${"0".repeat(64)}` })), LOCAL_QUOTES.map((q) => ({ ...q, chainId: "1" }))]) {
        const restricted = nativeOnly(quotes);
        const spy = [
          spyOn(Queries.prototype, "readZeroSeedList").mockImplementation((input) => restricted.list(input)),
          spyOn(Queries.prototype, "readZeroSeedDetail").mockImplementation((chain, pool) => restricted.detail(chain, pool)),
          spyOn(Queries.prototype, "readZeroSeedSwaps").mockImplementation((input) => restricted.swaps(input)),
          spyOn(Queries.prototype, "readZeroSeedClaims").mockImplementation((input) => restricted.claims(input)),
          spyOn(Queries.prototype, "readZeroSeedOwners").mockImplementation((input) => restricted.owners(input)),

        try {

          for (const label of ["l2", "l4"])

              const body = await get(`${path(label)}${suffix}`, 404);
              expect(body).toEqual({ status: 404, error: "Launch not found" });
            }
          expect(await get(path("l1"))).toEqual(detail);
        } finally {
          spy.forEach((s) => s.mockRestore());
        }
      }
      spies.forEach((s) => s.mockRestore());
      spies.splice(0, spies.length,
        spyOn(Queries.prototype, "readZeroSeedList").mockImplementation((input) => reader.list(input)),

        spyOn(Queries.prototype, "readZeroSeedSwaps").mockImplementation((input) => reader.swaps(input)),
        spyOn(Queries.prototype, "readZeroSeedClaims").mockImplementation((input) => reader.claims(input)),
        spyOn(Queries.prototype, "readZeroSeedOwners").mockImplementation((input) => reader.owners(input)),

      await get(path("l2"), 404);
      // Before the deployment block nothing is indexed: no coherent head.
      await invalidateFrom(db.dao, Number(before.created.block_number));
      await apiHead(db);
      await get(`/launches?chainId=${CHAIN_ID}`, 503);
      expect((await db.sql`SELECT COUNT(*)::int AS n FROM zero_seed_launch_states`)[0]!.n).toBe(0);
    } finally {
      spies.forEach((s) => s.mockRestore());
      await db.sql.end();
    }
